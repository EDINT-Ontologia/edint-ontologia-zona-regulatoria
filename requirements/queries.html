ZON01 - ¿Cuáles son las zonas reguladas?

Pregunta: ¿Cuáles son las zonas reguladas?

Respuesta: Lista de todas las zonas reguladas con su tipo

# --- Query 1 ---
PREFIX edintzone: <http://vocab.linkeddata.es/datosabiertos/def/common/zone#>
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>

SELECT ?zona
WHERE {
    ?zona rdf:type edintzone:RegulatedZone .
}

# --- Query 2 ---
PREFIX edintzone: <http://vocab.linkeddata.es/datosabiertos/def/common/zone#>
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?zona
WHERE {
    ?zona rdf:type edintzone:RegulatedZone ;
          edintzone:hasRegulatedZoneType ?tipo .
}
ORDER BY ?zona
ZON02 - ¿Cuál es el municipio de cada zona de bajas emisiones?

Pregunta: ¿Cuál es el municipio de cada zona de bajas emisiones?

Respuesta: Municipio donde se encuentra cada zona de bajas emisiones

# --- Query 1 ---
PREFIX edintkos-rztype: <http://vocab.linkeddata.es/datosabiertos/kos/edint/RegulatedZoneType/>
PREFIX edintzone: <http://vocab.linkeddata.es/datosabiertos/def/common/zone#>
PREFIX esadm: <http://vocab.linkeddata.es/datosabiertos/def/sector-publico/territorio#>
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>

SELECT ?zbe ?municipio
WHERE {
    ?zbe rdf:type edintzone:RegulatedZone ;
         edintzone:hasRegulatedZoneType edintkos-rztype:LowEmissionZone ;
         edintzone:locatedIn ?municipio .
}
ORDER BY ?zbe

# --- Query 2 ---
PREFIX edintkos-rztype: <http://vocab.linkeddata.es/datosabiertos/kos/edint/RegulatedZoneType/>
PREFIX edintzone: <http://vocab.linkeddata.es/datosabiertos/def/common/zone#>
PREFIX esadm: <http://vocab.linkeddata.es/datosabiertos/def/sector-publico/territorio#>
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?zbe ?municipio ?nombreMunicipio
WHERE {
    ?zbe rdf:type edintzone:RegulatedZone ;
         edintzone:hasRegulatedZoneType edintkos-rztype:LowEmissionZone ;
         edintzone:locatedIn ?municipio .
    ?municipio rdfs:label ?nombreMunicipio .
}
ORDER BY ?zbe ?nombreMunicipio

# --- Query 3 ---
PREFIX edintkos-rztype: <http://vocab.linkeddata.es/datosabiertos/kos/edint/RegulatedZoneType/>
PREFIX edintzone: <http://vocab.linkeddata.es/datosabiertos/def/common/zone#>

PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT ?zbe ?municipio ?nombreMunicipio
WHERE {
    ?zbe rdf:type edintzone:RegulatedZone ;
         edintzone:hasRegulatedZoneType edintkos-rztype:LowEmissionZone ;

ZON05 - ¿Cuáles son las zonas activas actualmente?

Pregunta: ¿Cuáles son las zonas activas actualmente?

Respuesta: Zonas que están activas en la fecha actual

# --- Query 1 ---
PREFIX edintzone: <http://vocab.linkeddata.es/datosabiertos/def/common/zone#>
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT DISTINCT ?zona ?tipo
WHERE {
    ?zona rdf:type edintzone:RegulatedZone ;
          edintzone:hasRegulatedZoneType ?tipo .
    FILTER NOT EXISTS { ?zona edintzone:startDate ?inicio }
    FILTER NOT EXISTS { ?zona edintzone:endDate ?fin }
}
ORDER BY ?zona

# --- Query 2 ---
PREFIX edintzone: <http://vocab.linkeddata.es/datosabiertos/def/common/zone#>
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>
PREFIX xsd: <http://www.w3.org/2001/XMLSchema#>

SELECT DISTINCT ?zona ?tipo ?inicio ?fin
WHERE {
    ?zona rdf:type edintzone:RegulatedZone ;
          edintzone:hasRegulatedZoneType ?tipo .
    OPTIONAL { ?zona edintzone:startDate ?inicio . }
    OPTIONAL { ?zona edintzone:endDate ?fin . }
    FILTER (
        (!BOUND(?inicio) || xsd:date(?inicio) <= xsd:date(NOW()))
        &&
        (!BOUND(?fin) || xsd:date(?fin) >= xsd:date(NOW()))
    )
}
ORDER BY ?zona
ZON06 - ¿Cuántas zonas reguladas hay por municipio?

Pregunta: ¿Cuántas zonas reguladas hay por municipio?

Respuesta: Cantidad de zonas reguladas agrupadas por municipio

# --- Query 1 ---
PREFIX edintzone: <http://vocab.linkeddata.es/datosabiertos/def/common/zone#>
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?municipio ?tipoZona (COUNT(?zona) AS ?numZonas)
WHERE {
    ?zona rdf:type edintzone:RegulatedZone ;
          edintzone:hasRegulatedZoneType ?tipoZona .
    ?zona edintzone:locatedIn ?municipio .
}
GROUP BY ?municipio ?tipoZona
ORDER BY ?municipio ?tipoZona

ZON11 - ¿Qué zonas comparten entidad gestora?

Pregunta: ¿Qué zonas comparten entidad gestora?

Respuesta: Zonas que comparten la misma entidad gestora

# --- Query 1 ---
PREFIX edintzone: <http://vocab.linkeddata.es/datosabiertos/def/common/zone#>
PREFIX rdf: <http://www.w3.org/1999/02/22-rdf-syntax-ns#>
PREFIX rdfs: <http://www.w3.org/2000/01/rdf-schema#>

SELECT ?entidadGestora (GROUP_CONCAT(STR(?zona); SEPARATOR=", ") AS ?zonas)
WHERE {
    ?zona rdf:type edintzone:RegulatedZone ;
          edintzone:hasRegulatedZoneType ?tipoZona .
    ?zona edintzone:managedBy ?entidadGestora .
}
GROUP BY ?entidadGestora
HAVING (COUNT(?zona) > 1)
ORDER BY ?entidadGestora
